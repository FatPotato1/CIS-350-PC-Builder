274    def update_part_dropdown(self, *_):
275        """
276        Updates the dropdown menus based on the user's selection.
277
278        Args:
279            *_: Placeholder for other arguments
280        Return:
281            None
282        """
283        component = self.component_type.get()
284        brand = self.brand_var.get()
285
286        # automatically update brand menu
287        if component == "CPU":
288            self.brand_menu["values"] = ["Intel", "AMD"]
289            if brand not in ["Intel", "AMD"]:
290                self.brand_var.set("")
291        elif component == "GPU":
292            self.brand_menu["values"] = ["NVIDIA", "AMD"]
293            if brand not in ["NVIDIA", "AMD"]:
294                self.brand_var.set("")
295        else:
296            self.brand_menu["values"] = []
297            self.brand_var.set("")
298
299        if component in ("CPU", "GPU") and brand:
300            self.build[f"{component}_Brand"] = brand
301            self.update_build_display()
302        # fill in part menu based on brand selection
303        parts = []
304        if component == "CPU":
305            if brand == "Intel":
306                parts = hardware.get("Intel CPUs", [])
307            elif brand == "AMD":
308                parts = hardware.get("AMD CPUs", [])
309        elif component == "GPU":
310            if brand == "NVIDIA":
311                parts = hardware.get("NVIDIA GPUs", [])
312            elif brand == "AMD":
313                parts = hardware.get("AMD GPUs", [])
314        elif component == "Motherboard":
315            if self.build.get("CPU_Brand"):
316                parts = hardware.get("Motherboards", {}).get(self.build["CPU_Brand"], [])
317            else:
318                parts = []
319        elif component == "RAM":
320            parts = hardware.get("RAM", [])
321        elif component == "Storage":
322            parts = hardware.get("Storage", [])
323        elif component == "PSU":
324            parts = hardware.get("PSU", [])
325        elif component == "Case":
326            parts = hardware.get("Cases", [])
327
328        self.part_menu["values"] = parts
329        if self.part_var.get() not in parts:
330            self.part_var.set("")

333    def update_build_display(self):
334        """
335        Updates the displayed build and total cost label in the UI.
336
337        Each component line shows the part name and its price where known.
338        The total cost label at the bottom sums all priced components.
339
340        Return:
341            None
342        """
343        self.build_list.config(state="normal")
344        self.build_list.delete("1.0", tk.END)
345
346        for component, value in self.build.items():
347            # skip internal brand-tracking keys from the display
348            if component.endswith("_Brand"):
349                continue
350            if value:
351                price = auto_generate.get_build_price({**self.build, "__single__": (component, value)})
352                # look up price for just this one part
353                part_price = self._get_part_price(component, value)
354                if part_price is not None:
355                    line = f"{component}: {value}  (${part_price:,})\n"
356                else:
357                    line = f"{component}: {value}\n"
358            else:
359                line = f"{component}: ---\n"
360
361            self.build_list.insert(tk.END, line)
362
363        self.build_list.config(state="disabled")
364
365        # update total cost label
366        total = auto_generate.get_build_price(self.build)
367        self.cost_label.config(text=f"Total Build Cost: ${total:,}")

369    def _get_part_price(self, component, part_name):
370        """
371        Look up the price of a single component by its build key and part name.
372
373        Args:
374            component (str): Build dict key (e.g. "GPU", "CPU", "RAM").
375            part_name (str): The part name string.
376
377        Returns:
378            int or None: Price in dollars, or None if not found.
379        """
380        import components as comp
381        if component == "GPU":
382            category = "NVIDIA GPUs" if self.build.get("GPU_Brand") == "NVIDIA" else "AMD GPUs"
383            return comp.prices.get(category, {}).get(part_name)
384        elif component == "CPU":
385            category = "Intel CPUs" if self.build.get("CPU_Brand") == "Intel" else "AMD CPUs"
386            return comp.prices.get(category, {}).get(part_name)
387        elif component == "RAM":
388            ram_prices = comp.prices.get("RAM", {})
389            for key, price in ram_prices.items():
390                if key in part_name:
391                    return price
392        elif component == "Motherboard":
393            return comp.prices.get("Motherboards", {}).get(part_name)
394        elif component == "Case":
395            return comp.prices.get("Cases", {}).get(part_name)
396        elif component == "Storage":
397            return comp.prices.get("Storage", {}).get(part_name)
398        elif component == "PSU":
399            return comp.prices.get("PSU", {}).get(part_name)
400        return None

460    def _on_algorithm_change(self, *_):
461        """
462        Show or hide algorithm-specific option panels when the selected algorithm changes.
463
464        Panel visibility rules:
465          "Minimum Cost" -> spec tier panel only
466          "Budget"       -> budget panel only (spec tier not shown; the budget cap

470        Args:

472        Return:
473            None
474        """
475        algo = self.algorithm_var.get()
476
477        # spec tier (min/rec radio buttons) — only meaningful for Minimum Cost
478        if algo == "Minimum Cost":
479            self.tier_frame.pack(fill=tk.X, padx=5, pady=2)

481            self.tier_frame.pack_forget()
482
483        # budget entry — only for Budget algorithm
484        if algo == "Budget":

486        else:
487            self.budget_frame.pack_forget()

489    # using this temporarily to generate build
490    def generate_build(self):
491        """
492        Generates a build based on the selected game and algorithm.
493
494        Dispatch table:
495          "Minimum Cost" -> generate_pc (original algorithm, respects spec tier)
496          "Budget"       -> generate_pc_budget (best balanced build within budget)
497
498        Return:
499             None
500        """
501        strategy = self.algorithm_var.get()
502        self.status_label.config(text="")
503
504        if strategy == "Budget":
505            budget = self._parse_budget()
506            if budget is None:
507                return
508            self.build["GPU"], self.build["CPU"], self.build["RAM"] = auto_generate.generate_pc_budget(
509                self.game_var.get(),
510                self.games,
511                self.cpu_rankings,
512                self.gpu_rankings,
513                self.build.get("CPU_Brand"),
514                self.build.get("GPU_Brand"),
515                budget,
516            )
517            if self.build["GPU"] is None or self.build["CPU"] is None or self.build["RAM"] is None:
518                self.status_label.config(
519                    text="budget is too low to create a build"
520                )
521
522        else:
523            # default: original Minimum Cost algorithm
524            self.build["GPU"], self.build["CPU"], self.build["RAM"] = auto_generate.generate_pc(
525                self.game_var.get(),
526                self.requirement_level.get(),
527                self.games,
528                self.cpu_rankings,
529                self.gpu_rankings,
530                self.build.get("CPU_Brand"),
531                self.build.get("GPU_Brand"),
532                strategy
533            )
534
535        # auto-populate fixed components for both algorithms
536        self._auto_populate_fixed_parts()
537
538        self.update_build_display()

540    def _auto_populate_fixed_parts(self):
541        """
542        Automatically fill in fixed components that are not user-selected during
543        auto-generation: Storage, PSU, Motherboard (matched to CPU brand), and
544        the cheapest available Case.
545
546        Only sets a component if it is not already manually chosen by the user,
547        so manual selections made before auto-generating are preserved.
548
549        Return:
550            None
551        """
552        import components as comp
553
554        # Storage, single fixed drive
555        if not self.build.get("Storage"):
556            storage_list = comp.hardware.get("Storage", [])
557            if storage_list:
558                self.build["Storage"] = storage_list[0]
559
560        # PSU, single fixed unit
561        if not self.build.get("PSU"):
562            psu_list = comp.hardware.get("PSU", [])
563            if psu_list:
564                self.build["PSU"] = psu_list[0]
565
566        # Motherboard, matched to CPU brand
567        if not self.build.get("Motherboard"):
568            cpu_brand = self.build.get("CPU_Brand", "Intel")
569            mobo_list = comp.hardware.get("Motherboards", {}).get(cpu_brand, [])
570            if mobo_list:
571                self.build["Motherboard"] = mobo_list[0]
572
573        # Case, cheapest available
574        if not self.build.get("Case"):
575            case_prices = comp.prices.get("Cases", {})
576            case_list = comp.hardware.get("Cases", [])
577            if case_prices and case_list:
578                cheapest = min(case_list, key=lambda c: case_prices.get(c, float("inf")))
579                self.build["Case"] = cheapest

581    def _parse_budget(self):
582        """
583        Parse and validate the budget entry field.
584
585        Returns:
586            float or None: The budget value, or None if the input is invalid
587            (error message written to status_label in that case).
588        """
589        try:
590            value = float(self.budget_var.get().replace(",", "").replace("$", "").strip())
591            if value <= 0:
592                raise ValueError
593            return value
594        except ValueError:
595            self.status_label.config(text="Not a valid budget.")
596            return None

598    def save_current_build(self):
599        """
600        Prompts the user to enter a name for the build. If the build doesn't have all info,
601        (missing a CPU or GPU), an error message is displayed and the save is cancelled.
602        If the user cancels or provides an empty name, the operation is also cancelled.
603
604        On successful save:
605            - The build is stored using the provided name.
606            - The saved builds list is refreshed.
607            - A status message is displayed to the user.
608
609        Returns:
610            None
611        """
612
613        # prevent saving empty builds
614        if not self.build.get("CPU") or not self.build.get("GPU"):
615            messagebox.showerror("Error", "Build is incomplete.")
616            return
617
618        # prompt user for name
619        name = simpledialog.askstring("Save Build", "Enter a name for this build:")
620
621        if not name or not name.strip():
622            return  # user cancelled or entered nothing
623
624        name = name.strip()
625
626        save_load.save_build(name, self.build.copy())
627        self.refresh_saved_builds()
628        self.status_label.config(text=f"Build '{name}' saved.")

630    def load_selected_build(self):
631        """
632        Load the currently selected build from saved builds.
633
634        If "None" is selected, the current build is cleared, including brand fields.
635        If no valid selection is made, an error message is shown.
636        And if the selected build exists, it replaces the current build with the loaded data.
637
638        Returns:
639            None
640        """
641        name = self.saved_builds_var.get()
642
643        if name == "None":
644            self.clear_build()
645            return
646
647            self.update_build_display()
648            self.status_label.config(text="Cleared current build.")
649            return
650
651        if not name:
652            messagebox.showerror("Error", "Select a build to load.")
653            return
654
655        loaded = save_load.load_build(name)
656
657        if not loaded:
658            messagebox.showerror("Error", "Build not found.")
659            return
660
661        self.build.clear()
662        self.build.update(loaded)
663        self.update_build_display()
664        self.status_label.config(text=f"Loaded '{name}'.")

666    def refresh_saved_builds(self):
667        """
668        Refresh the dropdown menu with all saved builds.
669
670        Brings all saved builds from storage and updates the dropdown menu

672        "None is set as default.

370    def _get_part_price(self, component, part_name):
371        """
372        Look up the price of a single component by its build key and part name.
373
374        Args:
375            component (str): Build dict key (e.g. "GPU", "CPU", "RAM").
376            part_name (str): The part name string.
377
378        Returns:
379            int or None: Price in dollars, or None if not found.
380        """
381        import components as comp
382        if component == "GPU":
383            category = "NVIDIA GPUs" if self.build.get("GPU_Brand") == "NVIDIA" else "AMD GPUs"
384            return comp.prices.get(category, {}).get(part_name)
385        elif component == "CPU":
386            category = "Intel CPUs" if self.build.get("CPU_Brand") == "Intel" else "AMD CPUs"
387            return comp.prices.get(category, {}).get(part_name)
388        elif component == "RAM":
389            ram_prices = comp.prices.get("RAM", {})
390            for key, price in ram_prices.items():
391                if key in part_name:
392                    return price
393        elif component == "Motherboard":
394            return comp.prices.get("Motherboards", {}).get(part_name)
395        elif component == "Case":
396            return comp.prices.get("Cases", {}).get(part_name)
397        elif component == "Storage":
398            return comp.prices.get("Storage", {}).get(part_name)
399        elif component == "PSU":
400            return comp.prices.get("PSU", {}).get(part_name)
401        return None

493        Generates a build based on the selected game and algorithm.

496          "Minimum Cost" -> generate_pc (original algorithm, respects spec tier)

519                self.status_label.config(
520                    text="budget is too low to create a build"

526                self.game_var.get(),
527                self.requirement_level.get(),

530                self.gpu_rankings,
531                self.build.get("CPU_Brand"),
532                self.build.get("GPU_Brand"),

541    def _auto_populate_fixed_parts(self):
542        """
543        Automatically fill in fixed components that are not user-selected during
544        auto-generation: Storage, PSU, Motherboard (matched to CPU brand), and
545        the cheapest available Case.
546
547        Only sets a component if it is not already manually chosen by the user,
548        so manual selections made before auto-generating are preserved.
549
550        Return:
551            None
552        """
553        import components as comp
554
555        # Storage, single fixed drive
556        if not self.build.get("Storage"):
557            storage_list = comp.hardware.get("Storage", [])
558            if storage_list:
559                self.build["Storage"] = storage_list[0]
560
561        # PSU, single fixed unit
562        if not self.build.get("PSU"):
563            psu_list = comp.hardware.get("PSU", [])
564            if psu_list:
565                self.build["PSU"] = psu_list[0]
566
567        # Motherboard, matched to CPU brand
568        if not self.build.get("Motherboard"):
569            cpu_brand = self.build.get("CPU_Brand", "Intel")
570            mobo_list = comp.hardware.get("Motherboards", {}).get(cpu_brand, [])
571            if mobo_list:
572                self.build["Motherboard"] = mobo_list[0]
573
574        # Case, cheapest available
575        if not self.build.get("Case"):
576            case_prices = comp.prices.get("Cases", {})
577            case_list = comp.hardware.get("Cases", [])
578            if case_prices and case_list:
579                cheapest = min(case_list, key=lambda c: case_prices.get(c, float("inf")))
580                self.build["Case"] = cheapest

492        """
